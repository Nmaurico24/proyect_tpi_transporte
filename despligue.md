### en la terminal ingresa

cd docker
### luego para rescontruir y levantar los contenedores

docker compose up --build -d



### para darle detener los contenedores ingresa

docker compose down




### para probar swagger iongresa en el navegador alguno de los links (pero solo el link, desde "http..")

ms-recursos 8081 http://localhost:8081/swagger-ui.html
ms-logistica 8082 http://localhost:8082/swagger-ui.html
ms-operaciones 8083 http://localhost:8083/swagger-ui.html
